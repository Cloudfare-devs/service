def handle_missing_values(data, numeric_method="mean"):
    """
    Handle numerical missing values using mean or median.
    Handle categorical missing values using mode.
    """

    result = data.copy()

    # Numerical columns
    for column in numeric_columns:

        if result[column].isnull().sum() > 0:

            if numeric_method == "mean":
                value = result[column].mean()

            else:
                value = result[column].median()

            result[column] = result[column].fillna(value)

    # Categorical columns
    for column in categorical_columns:

        if result[column].isnull().sum() > 0:

            mode_value = result[column].mode()[0]

            result[column] = result[column].fillna(
                mode_value
            )

    # Target missing values cannot be used for training
    result = result.dropna(
        subset=[TARGET]
    )

    return result

def detect_outliers_iqr(data):

    outlier_counts = {}

    for column in numeric_columns:

        Q1 = data[column].quantile(0.25)
        Q3 = data[column].quantile(0.75)

        IQR = Q3 - Q1

        lower_bound = Q1 - 1.5 * IQR
        upper_bound = Q3 + 1.5 * IQR

        outliers = data[
            (data[column] < lower_bound) |
            (data[column] > upper_bound)
        ]

        outlier_counts[column] = len(outliers)

    return outlier_counts

        }

        result[column] = (
            result[column]

        )

    return result, mappings

def train_test_split_manual(
    X,
    y,
    test_size=0.2,
    seed=42
):

    rng = np.random.default_rng(seed)

    indices = np.arange(len(X))

    rng.shuffle(indices)

    test_count = int(
        len(X) * test_size
    )

    test_indices = indices[
        :test_count
    ]

    train_indices = indices[
        test_count:
    ]

    X_train = X[
        train_indices
    ]

    X_test = X[
        test_indices
    ]

    y_train = y[
        train_indices
    ]

    y_test = y[
        test_indices
    ]

    return (
        X_train,
        X_test,
        y_train,
        y_test
    )

# MIN-MAX SCALING
# ------------------------------------------------------------

def min_max_fit_transform(
    X_train,
    X_test
):

    minimum = X_train.min(
        axis=0
    )

    maximum = X_train.max(
        axis=0
    )

    denominator = (
        maximum - minimum
    )

    denominator[
        denominator == 0
    ] = 1

    X_train_scaled = (
        X_train - minimum
    ) / denominator

    X_test_scaled = (
        X_test - minimum
    ) / denominator

    return (
        X_train_scaled,
        X_test_scaled
    )


# ------------------------------------------------------------
# Z-SCORE STANDARDIZATION
# ------------------------------------------------------------

def standardize_fit_transform(
    X_train,
    X_test
):

    mean = X_train.mean(
        axis=0
    )

    std = X_train.std(
        axis=0
    )

    std[
        std == 0
    ] = 1

    X_train_scaled = (
        X_train - mean
    ) / std

    X_test_scaled = (
        X_test - mean
    ) / std

    return (
        X_train_scaled,
        X_test_scaled
    )

# PART B - DISTANCE FUNCTIONS FROM SCRATCH

def euclidean_distance(
    a,
    b
):

    return np.sqrt(
        np.sum(
            (a - b) ** 2
        )
    )


def manhattan_distance(
    a,
    b
):

    return np.sum(
        np.abs(a - b)
    )


def minkowski_distance(
    a,
    b,
    p=3
):

    return (
        np.sum(
            np.abs(a - b) ** p
        )
    ) ** (1 / p)


def cosine_distance(
    a,
    b
):

    numerator = np.dot(
        a,
        b
    )

    denominator = (
        np.linalg.norm(a) *
        np.linalg.norm(b)
    )

    if denominator == 0:
        return 1.0

    cosine_similarity = (
        numerator / denominator
    )

    return 1 - cosine_similarity


def chebyshev_distance(
    a,
    b
):

    return np.max(
        np.abs(a - b)
    )

def calculate_distance(
    a,
    b,
    method="euclidean"
):

    if method == "euclidean":

        return euclidean_distance(
            a,
            b
        )

    elif method == "manhattan":

        return manhattan_distance(
            a,
            b
        )

    elif method == "minkowski":

        return minkowski_distance(
            a,
            b
        )

    elif method == "cosine":

        return cosine_distance(
            a,
            b
        )

    elif method == "chebyshev":

        return chebyshev_distance(
            a,
            b
        )

    else:

        raise ValueError(
            "Unknown distance method"
        )


# ============================================================
# NORMAL k-NN PREDICTION
# ============================================================

def knn_predict(
    X_train,
    y_train,
    X_test,
    k,
    distance_method="euclidean"
):

    predictions = []

    for test_sample in X_test:

        distances = []

        for i in range(
            len(X_train)
        ):

            distance = calculate_distance(
                test_sample,
                X_train[i],
                distance_method
            )

            distances.append(
                (distance, y_train[i])
            )

        # Sort by distance
        distances.sort(
            key=lambda x: x[0]
        )

        # Select k nearest neighbours
        neighbours = distances[:k]

        neighbour_labels = [
            label
            for distance, label
            in neighbours
        ]

        # Majority voting
        labels, counts = np.unique(
            neighbour_labels,
            return_counts=True
        )

        prediction = labels[
            np.argmax(counts)
        ]

        predictions.append(
            prediction
        )

    return np.array(
        predictions
    )


# ============================================================
# PART D - WEIGHTED k-NN
# ============================================================

def weighted_knn_predict(
    X_train,
    y_train,
    X_test,
    k,
    distance_method="euclidean"
):

    predictions = []

    for test_sample in X_test:

        distances = []

        for i in range(
            len(X_train)
        ):

            distance = calculate_distance(
                test_sample,
                X_train[i],
                distance_method
            )

            distances.append(
                (distance, y_train[i])
            )

        # Sort distances
        distances.sort(
            key=lambda x: x[0]
        )

        neighbours = distances[:k]

        class_weights = {}

        for distance, label in neighbours:

            weight = (
                1 /
                (distance + EPSILON)
            )

            if label not in class_weights:
                class_weights[label] = 0

            class_weights[label] += weight

        prediction = max(
            class_weights,
            key=class_weights.get
        )

        predictions.append(
            prediction
        )

    return np.array(
        predictions
    )

def accuracy_score_manual(
    y_true,
    y_pred
):

    correct = np.sum(
        y_true == y_pred
    )

    return correct / len(
        y_true
    )


def confusion_matrix_manual(
    y_true,
    y_pred
):

    labels = np.unique(
        np.concatenate(
            (y_true, y_pred)
        )
    )

    matrix = np.zeros(
        (len(labels), len(labels)),
        dtype=int
    )

    for actual, predicted in zip(
        y_true,
        y_pred
    ):

        actual_index = np.where(
            labels == actual
        )[0][0]

        predicted_index = np.where(
            labels == predicted
        )[0][0]

        matrix[
            actual_index,
            predicted_index
        ] += 1

    return labels, matrix


def precision_recall_f1(
    y_true,
    y_pred
):

    labels = np.unique(
        np.concatenate(
            (y_true, y_pred)
        )
    )

    precision_values = []
    recall_values = []
    f1_values = []

    for label in labels:

        TP = np.sum(
            (y_true == label) &
            (y_pred == label)
        )

        FP = np.sum(
            (y_true != label) &
            (y_pred == label)
        )

        FN = np.sum(
            (y_true == label) &
            (y_pred != label)
        )

        precision = (
            TP / (TP + FP)
            if (TP + FP) > 0
            else 0
        )

        recall = (
            TP / (TP + FN)
            if (TP + FN) > 0
            else 0
        )

        f1 = (
            2 * precision * recall /
            (precision + recall)
            if (precision + recall) > 0
            else 0
        )

        precision_values.append(
            precision
        )

        recall_values.append(
            recall
        )

        f1_values.append(
            f1
        )

    return (
        np.mean(precision_values),
        np.mean(recall_values),
        np.mean(f1_values)
    )
