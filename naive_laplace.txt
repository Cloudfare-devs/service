# 5. MULTINOMIAL NAIVE BAYES FROM SCRATCH
# ============================================================

class MultinomialNaiveBayes:

    def __init__(self, alpha):

        self.alpha = alpha

        self.classes = None

        self.class_prior = {}

        self.word_counts = {}

        self.total_words = {}

        self.word_probabilities = {}

        self.vocabulary_size = 0


    # --------------------------------------------------------
    # TRAIN MODEL
    # --------------------------------------------------------

    def fit(self, X, y):

        self.classes = np.unique(y)

        self.vocabulary_size = X.shape[1]

        total_documents = len(y)

        for current_class in self.classes:

            # Documents belonging to this class
            class_documents = X[
                y == current_class
            ]

            # Number of documents
            class_document_count = len(
                class_documents
            )

            # Prior probability
            prior = (
                class_document_count /
                total_documents
            )

            self.class_prior[
                current_class
            ] = prior


            # Count words in this class
            counts = np.sum(
                class_documents,
                axis=0
            )

            self.word_counts[
                current_class
            ] = counts

            # Total number of words
            total_words = np.sum(
                counts
            )

            self.total_words[
                current_class
            ] = total_words


            # ----------------------------------------------
            # Multinomial Naive Bayes formula
            #
            # P(w|C) =
            #
            # Count(w,C) + alpha
            # -----------------
            # N_C + alpha * V
            # ----------------------------------------------

            probabilities = (
                counts + self.alpha
            ) / (
                total_words +
                self.alpha *
                self.vocabulary_size
            )

            self.word_probabilities[
                current_class
            ] = probabilities

        return self


    # --------------------------------------------------------
    # PREDICT ONE MESSAGE
    # --------------------------------------------------------

    def predict_one(self, message_vector):

        class_scores = {}

        for current_class in self.classes:

            # Start with prior probability
            score = np.log(
                self.class_prior[
                    current_class
                ]
            )

            # Add word probabilities
            log_word_probabilities = np.log(
                self.word_probabilities[
                    current_class
                ]
            )

            score += np.sum(
                message_vector *
                log_word_probabilities
            )

            class_scores[
                current_class
            ] = score

        return max(
            class_scores,
            key=class_scores.get
        )


    # --------------------------------------------------------
    # PREDICT MULTIPLE MESSAGES
    # --------------------------------------------------------

    def predict(self, X):

        predictions = []

        for message in X:

            predictions.append(
                self.predict_one(message)
            )

        return np.array(
            predictions
        )


    # --------------------------------------------------------
    # PREDICT CLASS PROBABILITIES
    # --------------------------------------------------------

    def predict_proba(self, X):

        all_probabilities = []

        for message in X:

            log_scores = []

            for current_class in self.classes:

                log_prior = np.log(
                    self.class_prior[
                        current_class
                    ]
                )

                log_word_probability = np.log(
                    self.word_probabilities[
                        current_class
                    ]
                )

                score = (
                    log_prior +
                    np.sum(
                        message *
                        log_word_probability
                    )
                )

                log_scores.append(
                    score
                )


            # Convert log scores to probabilities
            log_scores = np.array(
                log_scores
            )

            max_score = np.max(
                log_scores
            )

            exp_scores = np.exp(
                log_scores - max_score
            )

            probabilities = (
                exp_scores /
                np.sum(exp_scores)
            )

            all_probabilities.append(
                probabilities
            )

        return np.array(
            all_probabilities
        )

# 13. EXPLAIN LAPLACE ESTIMATION
# ============================================================

print("\n" + "=" * 70)
print("EXPLANATION OF LAPLACE ESTIMATION")
print("=" * 70)

print("""
Multinomial Naive Bayes calculates the probability of a word
given a class.

Without smoothing:

P(w|C) = Count(w,C) / N_C

If a word has never appeared in a class:

Count(w,C) = 0

Therefore:

P(w|C) = 0

Naive Bayes multiplies the probabilities of words in a message.
Therefore, one zero probability can make the probability of
the entire message equal to zero.

Laplace estimation solves this problem by adding 1 to every
word count.

The formula becomes:

P(w|C) =
(Count(w,C) + 1)
-----------------
(N_C + V)

where:

Count(w,C) = number of times word w occurs in class C

N_C = total number of words in class C

V = total number of unique words in the vocabulary

For an unseen word:

Count(w,C) = 0

Therefore:

P(w|C) = 1 / (N_C + V)

The probability is now small but NOT zero.

Therefore, Laplace smoothing prevents the zero-probability
problem and allows the classifier to handle words that were
not observed in a particular class during training.
""")
